function fetchInventory() { fetch(GOOGLE_SHEET_CSV_URL) .then(response => response.text()) .then(csvText => { Papa.parse(csvText, { header: true, skipEmptyLines: true, complete: function(results) { itemsData = results.data.filter(row => row.Item && row.Item.trim() !== ''); renderItems(itemsData); } }); }) .catch(err => { console.error('Error fetching CSV:', err); document.getElementById('itemsGrid').innerHTML = `
Failed to load inventory. Please ensure your Google Sheet is published to the web.
`; }); }
